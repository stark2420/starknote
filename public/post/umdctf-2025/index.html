import re
from z3 import *

def parse_constraint(line, s):
    tokens = re.findall(r's\[\d+\]|\d+|==|!=|>=|<=|>|<', line)

    # s[...]や数字をZ3のオブジェクトに変換する
    def resolve(token):
        if token.startswith('s['):
            index = int(token[2:-1])
            return s[index]
        else:
            return int(token)

    conditions = []
    for i in range(0, len(tokens) - 2, 2):
        left = resolve(tokens[i])
        op = tokens[i + 1]
        right = resolve(tokens[i + 2])
        
        # Z3 の比較式を構築
        if op == '==':
            cond = left == right
        elif op == '!=':
            cond = left != right
        elif op == '>':
            cond = left > right
        elif op == '<':
            cond = left < right
        elif op == '>=':
            cond = left >= right
        elif op == '<=':
            cond = left <= right
        else:
            raise ValueError(f"Unknown operator: {op}")
        
        conditions.append(cond)

    return conditions
